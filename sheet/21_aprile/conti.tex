\section{Equations}
I want to obtain 
\begin{equation}
    C_l^{AB, obs}(k,k')
\end{equation}
when considering two fields $f_A(\vec{\textbf{r}})$ and $f_B(\vec{\textbf{r}})$, with $\vec{\textbf{r}} = (r, \theta, \phi)$. Much like what happens in \ref{eq12_sfb_bao}. \\
I use the SFB decomposition on fields $f_A(\vec{\textbf{r}})$ and $f_B(\vec{\textbf{r}})$ like in \ref{eq1_sfb_bao} and \ref{eq2_sfb_bao}:
\begin{equation}
    f_A(\vec{\textbf{r}}) = \sqrt{\frac{2}{\pi}} \sum_{l,m} \int_{0}^{\infty} dk f_{A,lm}(k) k j_l(kr) Y_{lm}(\theta, \phi)
\end{equation}
and the same for B; with coefficients:
\begin{equation}
    f_{A,lm}(k) = \sqrt{\frac{2}{\pi}} \sum_{l,m} \int d^3r f_A(\vec{\textbf{r}}) k j_l(kr) Y^*_{lm}(\theta, \phi)
\end{equation}   
and the same for B. But we do observe, like in \ref{eq8_sfb_bao}, that our surveys introduce:
\begin{equation}
    f_A^{obs}(\vec{\textbf{r}}) = \phi_A(\vec{\textbf{r}}) f_A(\vec{\textbf{r}})
\end{equation}
and the same for $f_B$. The selection function can be decomposed in:
\begin{equation}
    \phi_i(\vec{\textbf{r}}) = \phi_{i,r}(r) \phi_{i,t}(\theta, \phi)
\end{equation}
where the first factor is the radial selection function, and the other is the tangential selection function. \\
So we can write the SFB coefficients of $f_A^{obs}$ and $f_B^{obs}$ such:
\begin{equation}
    f_{A,lm}^{obs}(k) = \sqrt{\frac{2}{\pi}} \int d^3r \phi_A(\vec{\textbf{r}}) f_A(\vec{\textbf{r}}) k j_l(kr) Y^*_{lm}(\theta, \phi)
\end{equation}
From \ref{eq11_sfb_bao} we know that:
\begin{equation}
\langle f^{obs}_{A,lm}(k) f^{obs, *}_{B,l'm'}(k')\rangle = C^{AB,obs}_l(k,k') \delta_{ll'}\delta_{mm'}
\end{equation}
but we have to compute it directly (?). \\
Computing the two-point correlation between $f^{obs}_{A,lm}(k)$ and $f^{obs, *}_{B,l'm'}(k')$ we get:
\begin{equation}
    \langle f^{obs}_{A,lm}(k) f^{obs, *}_{B,l'm'}(k')\rangle = \frac{2}{\pi} \int d^3r_A d^3r_B \phi_A(\vec{\textbf{r}}_A) \phi_B(\vec{\textbf{r}}_B) 
    \langle f_{A}(\vec{\textbf{r}}_A) f_{B}(\vec{\textbf{r}}_B)\rangle k k' j_l(kr_A) j_{l'}(k'r_B) Y_{l,m}(\theta_A, \phi_A) Y_{l',m'}(\theta_B, \phi_B)
\end{equation}
where $d^3r = r^2dr d \Omega_{\hat{n}} = r^2dr d \Omega_{\theta, \phi} $. \\
We make the assumption of full-sky coverage, so $\phi_{i,t}(\theta, \phi) = 1$ and $\phi_i(\vec{\textbf{r}}) = \phi_{i}(r)$. 
The integral in $d\Omega_{\hat{n}}$ results in:
\begin{equation}
    \int d \Omega_{A} d \Omega_B Y^*_{l,m}(\theta_A, \phi_A) Y_{l',m'}(\theta_B, \phi_B) = \delta_{ll'} \delta_{mm'}
\end{equation}
and we are left with:
\begin{equation}
    \langle f^{obs}_{A,lm}(k) f^{obs, *}_{B,l'm'}(k')\rangle = \frac{2}{\pi} \int dr_A dr_B r^2_A r^2_B \phi_A(\vec{\textbf{r}}_A) \phi_B(\vec{\textbf{r}}_B) 
    \langle f_{A}(\vec{\textbf{r}}_A) f_{B}(\vec{\textbf{r}}_B)\rangle k k' j_l(kr_A) j_{l'}(k'r_B)
\end{equation}
How to deal with $\langle f_{A}(\vec{\textbf{r}}_A) f_{B}(\vec{\textbf{r}}_B)\rangle$? I tried to expand it into Fourier:
\begin{equation}
    \int \frac{d^3k''}{(2\pi)^3} P^{AB}(k'') e^{i\vec{\textbf{k}''\cdot(\vec{\textbf{r}}_A-\vec{\textbf{r}}_B)}}
\end{equation}
The integral in $d \Omega$ for the exponential gives (must be checked):
\begin{equation}
    \int d\Omega e^{i\vec{\textbf{k}''\cdot(\vec{\textbf{r}}_A-\vec{\textbf{r}}_B)}} = 4\pi i^{-l} j_l(k''r)Y_{lm}^{*}(\theta'',\phi'')
\end{equation}
then the two-point correlation function becomes:
\begin{equation}
    \langle f^{obs}_{A,lm}(k) f^{obs, *}_{B,l'm'}(k')\rangle = \frac{2}{\pi} \int dr_A r^2_A \phi_A(\vec{\textbf{r}}_A) j_l(kr_A) dr_B r^2_B \phi_B(\vec{\textbf{r}}_B) j_{l'}(k'r_B) k k' \int \frac{k''dk''}{(2\pi)^3} 4\pi P^{AB}(k'') j_l(k''r_A) j_l(k''r_B)
\end{equation}
by defining
\begin{equation}
    W_l^A(k,k'') = \int dr_A r^2_A \phi_A(\vec{\textbf{r}}_A) j_l(kr_A) j_l(k''r_A)
\end{equation}
and 
\begin{equation}
    W_l^B(k,k'') = \int dr_B r^2_B \phi_B(\vec{\textbf{r}}_B) j_l(kr_B) j_l(k''r_B)    
\end{equation}
we get 
\begin{equation}
    C_l^{AB,obs}(k,k') = (\frac{2}{\pi})^2 \frac{4\pi}{(2\pi)^3} \int dk'' k''^2 P^{AB}(k'') W_l^A(k,k'') W_l^B(k',k'')
\end{equation}(the coefficients surely must be checked.
