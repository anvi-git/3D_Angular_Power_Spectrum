\section{Stef's eqs}
In the linear regime, the galaxy clustering harmonic-space power spectrum is
\begin{equation}
    C_l^{\rm gg}(r_1,r_2) = \frac{2}{\pi}\,W^{\rm g}(r_1)\,W^{\rm g}(r_2)\,\int_0^{\infty}\de k\,k^2\,P_{\rm lin}(k)\,j_l(k\,r_1)\,j_l(k\,r_2)\;,
% \label{eq15_blast}    
\end{equation}
with
\begin{equation}
    W^{\rm g}(r)=b(r)\,n(r)\,D(r)\;.
\end{equation}
To obtain the sFB power spectrum, we need to perform a Hankel transform for each line of sight, resulting into a new kernel in Fourier space,
\begin{equation}
    \widetilde W^{\rm g}_l(k)=4\,\pi\,\int_0^{\infty}\de r\,r^2\,W^{\rm g}(r)\,j_l(k\,r)\;.
\end{equation}
Thus, the sFB power spectrum reads
\begin{equation}
    S_l^{\rm gg}(k_1,k_2) = \frac{2}{\pi}\,\int_0^{\infty} \de r_1\,W^{\rm g}(r_1)\,\int_0^{\infty}\de r_2\,W^{\rm g}(r_2)\,\int_0^{\infty}\de k\,k^2\,P_{\rm lin}(k)\,j_l(k\,r_1)\,j_l(k\,r_2)\;,
% \label{eq15_blast}    
\end{equation}
