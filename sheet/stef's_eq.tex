\section{Stef's eqs}
In the linear regime, the harmonic-space power spectrum for observables \(A\) and \(B\) is
\begin{equation}
    C_l^{AB}(z_1,z_2) = \frac{2}{\pi}\,\int_0^{\infty}\de k\,k^2\,P_{\rm lin}(k)W^A_l(k,z_1)\,W^B_l(k,z_2)\;.
% \label{eq15_blast}    
\end{equation}
To obtain the sFB power spectrum, we need to perform a Hankel transform for each line of sight, resulting into a new kernel in Fourier space,
\begin{equation}
    \widetilde W^A_l(k,q)=4\,\pi\,\int_0^{\infty}\de r\,r^2\,W^A(k,r)\,j_l(q\,r)\;.
\end{equation}
Thus, the sFB power spectrum reads
\begin{equation}
    S_l^{AB}(k_1,k_2) = \frac{2}{\pi}\,\int_0^{\infty}\de q\,q^2\,P_{\rm lin}(q)\,\widetilde W^A_l(k_1,q)\,\widetilde W^B_l(k_2,q)\;.
% \label{eq15_blast}    
\end{equation}

For the various observables, we have
\begin{align}
\label{obs}
    W^{\rm den}_l(k,z)&=\frac{H(z)}c\,b(z)\,n(z)\,D(z)\,j_l[k\,r(z)]\;,\\
    W^\gamma_l(k,z)&=-\sqrt{\frac{(l+2)!}{(l-2)!}}\,\frac32\,H_0^2\,\Omega_{{\rm m},0}\,(1+z)\,q_{3/5}(z)\,D(z)\,\frac{j_l[k\,r(z)]}{k^2\,r(z)}\;,\\
    W^{\rm RSD}_l(k,z)&=\frac{H(z)}c\,f(z)\,n(z)\,D(z)\,\left\{\left[\frac{l\,(l-1)}{k^2\,r^2(z)}-1\right]\,j_l[k\,r(z)]+\frac2{k\,r(z)}\,j_{l+1}[k\,r(z)]\right\}\nonumber\\
    &\equiv\frac{f(z)}{b(z)}\left\{\left[\frac{l\,(l-1)}{k^2\,r^2(z)}-1\right]\,W^{\rm den}_l(k,z)+\frac2{k\,r(z)}\,W^{\rm den}_{l+1}(k,z)\right\}\;,\\
    W^{\rm mag}_l(k,z)&=-\sqrt{\frac{(l+2)!}{(l-2)!}}\,3\,H_0^2\,\Omega_{{\rm m},0}\,(1+z)\,q_s(z)\,D(z)\,\frac{j_l[k\,r(z)]}{k^2\,r(z)}\nonumber\\
    &\equiv2\,\frac{q_s(z)}{q_{3/5}(z)}\,W^\gamma_l(k,z)\;,
\end{align}
where, given \(s\) the magnification bias, \(q_s(z)\) is the magnification-bias weighted lensing efficiency function,
\begin{equation}
    q_s(z)=\int_r^\infty\de x\,\frac{x-r}x\,n(x)\,[5\,s(x)-2]\;.
\end{equation}
